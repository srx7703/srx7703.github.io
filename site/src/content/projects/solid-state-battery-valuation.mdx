// "Announced capacity" has to mean a capacity figure in gigawatt- or megawatt-hours. A row that says of
// itself that it is a licence ceiling, a readiness rating, a target year or a tonnage of material goes
// in the table below the capacity one instead, with its own caveat rendered beside it.
export const capacityRows = facts.capacity ?? [];
// Wording that says the row is not a capacity figure. Every one of these is quoted from the row's own
// `metric` or `unit` string: a licence ceiling, a readiness rating and a target year all say so of
// themselves. publish.py's own flag wins if it ever sets one; otherwise a doubtful row goes to the
// claims table, which is the safe direction to be wrong in.
export const NOT_CAPACITY = /n(?:o|ot)\s+(?:a\s+)?GWh|none disclosed|ceiling|licen[cs]e|target year|TRL|readiness|NOT built/i;
export const isCapacity = (c) => c.is_capacity ?? (
  /\b[GM]Wh\b/i.test(String(c.unit ?? '')) && !NOT_CAPACITY.test(`${c.unit ?? ''} ${c.metric ?? ''}`)
);
export const realCapacity = capacityRows.filter(isCapacity);
export const claimRows = capacityRows.filter((c) => !isCapacity(c));

  .map((c) => c.ticker);
// Price to sales only where the trading currency and the reporting currency agree: a market value in
// one currency over revenue in another is not a ratio. publish.py's own field wins if it exists.
export const priceToSales = (p) => p.price_to_sales ?? (
  p.market_cap != null && p.ttm_revenue > 0 && p.price_currency === p.reporting_currency
    ? p.market_cap / p.ttm_revenue
    : null
);
export const mixedCurrency = pure.filter((p) => p.price_currency !== p.reporting_currency).length;

export const evalStatus = (key) => {
  const r = scored[key];
  if (!r) return 'Not scored yet — no scorer output was on disk when this page was built.';
  if (r.status === 'not_yet') return `Not yet: ${r.why}`;
  const parts = [];
  if (key === 'valuation_dispersion' && r.p25 != null) parts.push(`quartiles ${x1(r.p25)} to ${x1(r.p75)} across ${r.n_priced} of ${r.n_listings} listings`);
  if (key === 'revision_persistence' && r.n_transitions != null) parts.push(`${r.n_transitions} transitions over ${r.periods_used} periods`);
  if (key === 'dual_listing_premium' && r.median_premium != null) parts.push(`median A line over H line ${pct(r.median_premium, 1)} across ${r.pairs.length} pairs`);
  if (key === 'share_bases') {
    if (r.rank_correlation) parts.push(`rank correlation ${r.rank_correlation.spearman.toFixed(2)} over ${r.rank_correlation.n} companies, against ${r.rank_correlation.source_name}`);
    else if (r.rank_correlation_why) parts.push(`no rank correlation yet: ${r.rank_correlation_why}`);
    if (r.share_difference) parts.push(`${r.share_difference.n} companies quoted a share, median gap ${pct(r.share_difference.median_abs_difference, 1)}`);
    else if (r.share_difference_why) parts.push(`no share comparison yet: ${r.share_difference_why}`);
  }
  if (r.note) parts.push(r.note);
  return `Reports now: ${parts.join('; ') || 'scored'}`;
};

<Section id="method" title="Method">
  <div class="prose">
    <p><strong>Question.</strong> What is the listed solid-state battery world being priced at, and how much of that price rests on a product that is not being sold in volume yet?</p>
    <p><strong>Pool.</strong> {facts.n_listings} listings across {facts.n_issuers} companies in mainland China, the United States, Japan, Korea, Hong Kong and the United Kingdom, in three groups: the pure plays, the cell makers with a solid-state roadmap, and the materials and equipment suppliers. Private developers are named in the shipments and capacity tables but cannot be valued, so they are not in the pool.</p>
    <p><strong>Prices and consensus.</strong> As on the optical page: Yahoo Finance for prices and non-Chinese consensus, East Money's per-broker forecasts for the A-shares, SEC XBRL and East Money quarterly reports and Yahoo for trailing earnings, FRED for exchange rates. Fiscal years are restated onto calendar years by month overlap, which matters here for the {offDecember} listings whose year does not end in December, including the Japanese names, whose years end in March.</p>
    <p><strong>Definitions.</strong> Trailing PE is market capitalisation over trailing-twelve-month attributable profit. Forward PE is price over calendar-year consensus EPS, with the consensus converted into the price currency first, and on the accounting basis of whoever published it — see the note above the table. Growth is the consensus for {y1} over the consensus for {y0}. Forecast spread is the gap between the highest and lowest forecast as a share of the mean, blended across fiscal years on the same weights as the mean itself, and blank wherever any contributing year is already reported, because a reported year has no range: {nSpread} of {facts.n_listings} listings carry one. Price to sales is market value over trailing revenue, shown only for the pure plays and only where there is revenue and the two currencies agree.</p>
    <p><strong>What is deliberately missing.</strong> No solid-state market share, because a market share of a market that does not exist is not a number. No negative PE: a company forecast to lose money gets a written reason instead. No estimate of how much of any company's value is "the solid-state option", which would be an opinion rather than a measurement. No enterprise value or cash runway for the pure plays yet, which is the gap this page most wants to close.</p>
  </div>
</Section>
